    bool profitableRecoveryFailureNow = false
    if recoveryTradeWatchEligible and recordedTradeOpen and strategy.position_size != 0 and isOneMinute and inSession and barstate.isconfirmed and not na(recordedEntryBar) and bar_index > recordedEntryBar
        bool recoveryAcceptanceClose = recordedTradeDirection == 1 ? close > recoveryTradeFpiTop : close < recoveryTradeFpiBottom
        if not recoveryTradeAcceptanceSeen
            if recoveryAcceptanceClose
                recoveryTradeAcceptanceSeen := true
                recoveryTradeAcceptanceBar := bar_index
        else if not na(recoveryTradeAcceptanceBar)
            int candlesAfterRecovery = bar_index - recoveryTradeAcceptanceBar
            if candlesAfterRecovery > FAILED_RECOVERY_CONFIRMATION_BARS
                recoveryTradeWatchEligible := false
            else if candlesAfterRecovery >= 1
                bool recoveryFarEdgeFailure = recordedTradeDirection == 1 ? close < recoveryTradeFpiBottom : close > recoveryTradeFpiTop
                profitableRecoveryFailureNow := recordedMaxProfitPoints >= FAILED_RECOVERY_PROFIT_POINTS and recoveryFarEdgeFailure

    // Count distinct touch episodes of the currently authoritative FPI. Consecutive bars
    // inside the zone are one retest. The second distinct retest forces a flat state first.
    bool activeFpiSecondRetestExitNow = false
    if not sessionConfluenceHoldActive and activeFpiValid and activeSourceKind != SOURCE_NONE and activeSourceKind != SOURCE_BGOB and activeSourceKind != SOURCE_ORDER_BLOCK and activeSourceKind != SOURCE_REVERSAL_FOUNDATION and not activeSecondRetestConsumed and not na(activeFpiBar) and bar_index > activeFpiBar
        bool activeZoneTouch = low <= activeTop and high >= activeBottom
        if activeZoneTouch and not activeTouchedLast
            activeRetestCount += 1
            if activeRetestCount >= 2 and recordedTradeOpen and strategy.position_size != 0
                activeFpiSecondRetestExitNow := true
                activeSecondRetestConsumed := true
        activeTouchedLast := activeZoneTouch

    // A trade whose original entry authority was a BGOB cannot remain open after price
    // penetrates beyond that BGOB's 50% midpoint.
    bool bgobMidpointExitNow = false
    if recordedTradeOpen and recordedOriginKind == SOURCE_BGOB and not na(recordedOriginTop) and not na(recordedOriginBottom) and not na(recordedEntryBar) and bar_index > recordedEntryBar
        float originBgobMidpoint = (recordedOriginTop + recordedOriginBottom) / 2.0
        bgobMidpointExitNow := recordedTradeDirection == 1 ? low < originBgobMidpoint : high > originBgobMidpoint

    // Requalification uses the same strict FPI sequence: accept outside in the original
    // direction, then physically retest on a later bar and close outside again. A close
    // through the far edge confirms the opposing direction without an original-bias entry.
    bool fpiRequalificationSignalNow = false
    int fpiRequalificationSignalDirection = 0
    if fpiRequalificationWatch and not targetWindowLock and not recordedTradeOpen and strategy.position_size == 0 and not na(fpiRequalificationBar) and bar_index > fpiRequalificationBar
        bool requalificationZoneRetest = low <= fpiRequalificationTop and high >= fpiRequalificationBottom
        bool requalificationAcceptanceClose = fpiRequalificationDirection == 1 ? close > fpiRequalificationTop : close < fpiRequalificationBottom
        bool requalificationFarEdgeFailure = fpiRequalificationDirection == 1 ? close < fpiRequalificationBottom : close > fpiRequalificationTop
        if requalificationFarEdgeFailure
            fpiRequalificationSignalNow := true
            fpiRequalificationSignalDirection := -fpiRequalificationDirection
            fpiRequalificationAcceptanceConfirmed := false
            fpiRequalificationAcceptanceBar := na
        else if not fpiRequalificationAcceptanceConfirmed
            if requalificationAcceptanceClose
                fpiRequalificationAcceptanceConfirmed := true
                fpiRequalificationAcceptanceBar := bar_index
        else if not na(fpiRequalificationAcceptanceBar) and bar_index > fpiRequalificationAcceptanceBar and requalificationZoneRetest and requalificationAcceptanceClose
            fpiRequalificationSignalNow := true
            fpiRequalificationSignalDirection := fpiRequalificationDirection
            fpiRequalificationAcceptanceConfirmed := false
            fpiRequalificationAcceptanceBar := na

    // If the opening FPI invoked its close-and-wait rule, a previously validated
    // retained foundation may restore the permanent session direction at 10:00,
    // but only while price is confirmed outside that foundation in its direction.
    bool openingConfluenceReleaseSignalNow = openingFpiLockReleasedNow and sessionConfluenceBiasActive and not recordedTradeOpen and strategy.position_size == 0 and not na(sessionConfluenceMasterTop) and not na(sessionConfluenceMasterBottom) and (sessionAnchorDirection == 1 ? close > sessionConfluenceMasterTop : close < sessionConfluenceMasterBottom)

    // A previously delayed profitable-trade reversal fires only after a confirmed close
    // through the frozen last swing-close level from before the contradicting period began.
    bool delayedFpiSignalNow = delayedFpiReversal and not targetWindowLock and not na(delayedFpiArmedBar) and bar_index > delayedFpiArmedBar and not na(delayedFpiBreakLevel) and (delayedFpiDirection == 1 ? close > delayedFpiBreakLevel : close < delayedFpiBreakLevel)

    if sessionConfluenceBiasActive and delayedFpiReversal and delayedFpiDirection != sessionAnchorDirection
        delayedFpiReversal := false
        delayedFpiSignalNow := false
    else if macro950GateActive and delayedFpiReversal and delayedFpiDirection != macro950AuthorityDirection
        delayedFpiReversal := false
        delayedFpiSignalNow := false
    if sessionRegimeGateActive and delayedFpiReversal and delayedFpiDirection != sessionExpectedDirection
        delayedFpiReversal := false
        delayedFpiSignalNow := false

    // If a trade has already achieved more than 50 points and a later regular period FPI
    // contradicts it, arm the frozen swing-level watch. The special 9:50 authority is
    // deliberately exempt: its boundary change acts immediately, as marked in the review.
    int incomingFpiDirection = mainSignalDirection
    float incomingFpiTop = mainSignalTop
    float incomingFpiBottom = mainSignalBottom
    int incomingFpiWindow = mainSignalWindow
    bool incomingFpiSignalNow = mainSignalNow
    bool incomingFpiOpposesTrade = incomingFpiSignalNow and recordedTradeOpen and strategy.position_size != 0 and incomingFpiDirection != positionDirection()
    bool incomingFpiAlignsTrade = incomingFpiSignalNow and recordedTradeOpen and strategy.position_size != 0 and incomingFpiDirection == positionDirection()
    bool incomingFpiFromLaterWindow = recordedWindow >= 0 and currentWindow >= 0 and currentWindow != recordedWindow
    float incomingFrozenBreakLevel = incomingFpiDirection == 1 ? frozenPriorSwingHighClose : frozenPriorSwingLowClose
    bool incomingFpiAllowedByMacro950 = sessionConfluenceBiasActive ? incomingFpiDirection == sessionAnchorDirection : not macro950GateActive or incomingFpiDirection == macro950AuthorityDirection
    bool incomingFpiAllowedBySession = sessionConfluenceBiasActive ? incomingFpiDirection == sessionAnchorDirection : not sessionRegimeGateActive or incomingFpiDirection == sessionExpectedDirection
    bool sessionAlignmentAuthorityNow = sessionAlignmentProofNow or intervalFoundationSignalNow
    bool delayIncomingFpiNow = not sessionAlignmentAuthorityNow and not bgobSignalNow and incomingFpiAllowedByMacro950 and incomingFpiAllowedBySession and incomingFpiOpposesTrade and incomingFpiFromLaterWindow and recordedMaxProfitPoints > PROFITABLE_HOLD_THRESHOLD_POINTS and not na(incomingFrozenBreakLevel)
